À propos de la page

Signaler une erreur

On met en place un système qui permet au tour de progresser de niveau. La mise à niveau demande une animation pour indiquer visuellement qu'il se passe quelque chose. Une approche simpliste serait de simplement changer le gameObject de la tour lorsqu'elle est changée. Toutefois, cette méthode perd la référence sur le script de la tour et complique la gestion des états. Pour cette raison, nous allons conserver le même GameObject et simplement modifier quelques valeurs pour refléter le nouveau niveau de la tour.

Dans la prochaine section, nous verrons comment animer les éléments visuels de la tour lors de son amélioration.

Structurer les données de la tour

Pour mieux gérer les différents niveaux de la tour, il est utile de structurer les données associées à chaque niveau. Une approche consiste à utiliser des Scriptable Objects pour définir les caractéristiques de chaque niveau, comme les dégâts, la portée et le coût d'amélioration. Cela permet de séparer les données de la logique et de faciliter les modifications sans toucher au code. Cela nous permettra aussi de changer le modèle utilisé pour représenter la tour.

Nous ferons donc un type de donnée NiveauTour sous forme de Scriptable Object pour représenter les caractéristiques de chaque niveau de la tour. Une tour sera responsable d'appliquer ce modèle lors d'une amélioration. Pour la représentation visuelle, nous pourrons changer le Mesh de la tour en fonction du niveau. Les materials sont identiques, donc nous n'avons pas à les modifier, mais eux aussi pourraient être changés.

/// <summary>
/// Représente un ScriptableObject de configuration pour un niveau de tour.
/// </summary>
using UnityEngine;

[CreateAssetMenu(fileName = "NiveauTour", menuName = "ABC/NiveauTour")]
public class NiveauTour : ScriptableObject
{
    [Header("Caractéristiques de la tour")]
    [field: SerializeField, Tooltip("Niveau de la tour")]
    public int Niveau { get; private set; }

    [field: SerializeField, Tooltip("Dégâts infligés par la tour")]
    public float Degats { get; private set; }

    [field: SerializeField, Tooltip("Délai entre deux attaques de la tour")]
    public float DelaiAttaque { get; private set; }

    [Header("Affichage de la tour")]
    [field: SerializeField, Tooltip("Mesh pour la tour")]
    public Mesh MeshTour { get; private set; }
}

Pour intégrer les niveaux dans la tour, il est nécessaire de stocker une référence au Scriptable Object correspondant au niveau actuel. La tour pourra ainsi accéder aux caractéristiques définies dans le NiveauTour et les appliquer lors de ses actions, comme l'attaque ou l'amélioration.

Les attributs "Niveau", "Dégâts" et "DelaiAttaque" de la tour seront obtenus à partir du Scriptable Object NiveauTour correspondant au niveau actuel. De cette manière, toute modification des caractéristiques d'un niveau dans le Scriptable Object sera automatiquement reflétée dans la tour.

La machine à états

Le changement de niveau correspondra à un état de la machine à état. Lorsque le changement est fait, la tour revient dans son état attaque. À la fin de l'état, le nouveau niveau est appliqué. On ajoute également un niveau maximal comme constante de la tour. La transition vers l'état d'amélioration ne sera possible que si le niveau actuel est inférieur au niveau maximal et si le joueur appuie sur le bouton d'amélioration.

public class Tour : MonoBehaviour
{
    // Niveau maximum que la tour peut atteindre
    public const int NIVEAU_MAX = 3;

    // Indique si une amélioration de la tour a été demandée par le joueur
    public bool AmeliorationDemandee { get; private set; }

    /// <summary>
    /// Niveau vers lequel l'amélioration est réalisée
    /// </summary>
    public NiveauTour NiveauAmelioreVers { get; private set; }

    // Compteur du temps pour appliquer l'amélioration de la tour.
    public float TempsAmelioration { get; set; }

    /// <summary>
    /// Demande d'appliquer une amélioration. Cette propriété est lu par la machine à état.
    /// </summary>
    public void Ameliorer()
    {
        if(niveauTour.Niveau < NIVEAU_MAX)
        {
            // TODO : dépenser des resources....

            AmeliorationDemandee = true;
            NiveauAmelioreVers = GetProchainNiveau();
        }
    }

    /// <summary>
    /// Applique les données du niveau amélioré vers à la tour et réinitialise le niveau amélioré vers null.
    /// </summary>
    public void AppliquerNiveau()
    {
        if(NiveauAmelioreVers == null)
        {
            Debug.LogError($"NiveauAmelioreVers est null. Impossible d'appliquer le niveau à la tour {NomTour}");
            return;
        }

        niveauTour = NiveauAmelioreVers;
        NiveauAmelioreVers = null;
        GetComponent<MeshFilter>().mesh = niveauTour.MeshTour;
    }

    /// <summary>
    /// Détermine le prochain niveau de tour.
    /// </summary>
    /// <returns>Le niveau de tour suivant.</returns>
    private NiveauTour GetProchainNiveau()
    {
        NiveauTour prochainNiveau = Resources.Load<NiveauTour>($"Niveaux/NiveauTour{Niveau + 1}");

        if (prochainNiveau == null)
        {
            Debug.LogWarning($"Aucun NiveauTour trouvé pour le niveau {Niveau + 1}. La tour est au niveau maximum.");
        }
        return prochainNiveau;
    }

public class EtatAttaque : EtatTour
{
    public override EtatTour OnExecuter(Tour tour)
    {
        if (tour.AmeliorationDemandee)
        {
            return new EtatAmelioration();
        }

        if (tour.TempsDepuisDerniereAttaque >= tour.DelaiAttaque
            && tour.CibleActive != null)
        {
            tour.Attaquer();
            tour.TempsDepuisDerniereAttaque = 0f;
        }

        tour.TempsDepuisDerniereAttaque += Time.deltaTime;

        return this;
    }
}

using UnityEngine;

namespace EtatsTour
{
    /// <summary>
    /// Etat de la tour lorsqu'elle est en train d'être améliorée
    /// </summary>
    public class EtatAmelioration : EtatTour
    {
        public override void OnEntrer(Tour tour)
        {
            base.OnEntrer(tour);

            tour.TempsAmelioration = 0f;
        }

        public override EtatTour OnExecuter(Tour tour)
        {
            if (tour.TempsAmelioration < tour.NiveauAmelioreVers.TempsAppliquerAmelioration)
            {
                tour.TempsAmelioration += Time.deltaTime;
                return this;
            }

            return new EtatAttaque();
        }

        public override void OnSortir(Tour tour)
        {
            base.OnSortir(tour);

            tour.AppliquerNiveau();
            tour.AmeliorationDemandee = false;
        }
    }
}

using UnityEngine;
using TMPro;
using UnityEngine.UI;

/// <summary>
/// Affiche une tour dans l'interface
/// </summary>
public class AfficheurTour : MonoBehaviour
{
    [SerializeField, Tooltip("Zone d'affichage de la sélection")]
    private GameObject zoneAffichage;

    [SerializeField, Tooltip("Référence au texte affichant le nom de la tour")]
    TextMeshProUGUI nomTourTexte;

    [SerializeField, Tooltip("Référence au texte affiché dans le bouton d'amélioration")]
    TextMeshProUGUI boutonAmeliorationTexte;

    // Bouton d'amélioration de la tour
    Button boutonAmelioration;

    // Référence à la tour actuellement affichée
    private Tour tourAffichee;

    private void Start()
    {
        ControleurJeu.Instance.GestionnaireSourisParDefaut.ObjetSelectionne += OnObjetSelectionne;

        // On sait que le parent est un bouton, ce code est habituellement risqué à utiliser
        boutonAmelioration = boutonAmeliorationTexte.transform.parent.GetComponent<Button>();
    }

    /// <summary>
    /// Réagit au changement d'objet sélectionné
    /// </summary>
    /// <param name="_">Le vieille objet sélectionné</param>
    /// <param name="nouvelleSelection">Le nouvel objet sélectionné</param>
    public void OnObjetSelectionne(GameObject _, GameObject nouvelleSelection)
    {
        if(nouvelleSelection != null && nouvelleSelection.TryGetComponent(out Tour tour))
        {
            zoneAffichage.SetActive(true);
            AfficherTour(tour);
            tourAffichee = tour;
        }
        else
        {
            zoneAffichage.SetActive(false);            
            tourAffichee = null;
        }
    }

    /// <summary>
    /// Lance l'amélioration de la tour affichée
    /// </summary>
    public void Ameliorer()
    {
        if (tourAffichee != null)
        {
            tourAffichee.Ameliorer();
        }
    }

    /// <summary>
    /// Affiche le contenu d'une tour
    /// </summary>
    /// <param name="tour">La tour à afficher</param>
    private void AfficherTour(Tour tour)
    {
        // Met à jour le texte du bouton d'amélioration en fonction du niveau de la tour
        if (tour.Niveau >= Tour.NIVEAU_MAX)
        {
            boutonAmeliorationTexte.text = "Niveau max";
            boutonAmelioration.interactable = false;
        }
        else
        {
            boutonAmeliorationTexte.text = $"Améliorer au niveau {tour.Niveau + 1}";
            boutonAmelioration.interactable = true;
        }

        // Met à jour le texte du nom de la tour
        nomTourTexte.text = tour != null ? tour.NomTour : "NULL";
    }
}
