Création des éléments¶

Dans ce système, nous développerons des éléments pour les tours et les ennemis. Chaque élément aura une faiblesse pour l'élément opposé (feu <-> eau, terre <-> air) et une immunité contre lui. Nous créerons des Scriptable Objects pour chaque élément afin de réutiliser facilement ces informations dans le projet.

Chaque élément aura aussi un material associé pour représenter visuellement l'élément.

using UnityEngine;

/// <summary>
/// Type d'élément utilisé par les tours et les ennemis. Les éléments ont un material et une couleur associée.
/// </summary>
[CreateAssetMenu(fileName = "Element", menuName = "ABC/Element")]
public class Element : ScriptableObject
{
    [SerializeField, Tooltip("Nom de l'élément")]
    private string nom;

    [field: SerializeField, Tooltip("Matériau de l'élément")]
    public Material Material { get; private set; }

    /// <summary>
    /// Retourne la couleur de l'élément
    /// </summary>
    public Color Couleur
    {
        get
        {
            if (Material != null)
            {
                return Material.color;
            }
            else
            {
                Debug.LogWarning($"Le matériau de l'élément '{nom}' n'est pas défini.");
                return Color.white; // Retourne une couleur par défaut si le matériau n'est pas défini
            }
        }
    }

    [SerializeField, Tooltip("Élément contre lequel cet élément est efficace")]
    private Element vulnerabiliteA;

    /// <summary>
    /// Calcule le multiplicateur de dégâts appliqué selon l’élément reçu.
    /// </summary>
    /// <param name="element">Élément utilisé pour déterminer l’efficacité des dégâts.</param>
    /// <returns>Le facteur multiplicatif des degat inflige à un ennemi qui possède cet élément</returns>
    public float GetMultiplicateurDegat(Element element)
    {
        if (element == vulnerabiliteA)
        {
            return 2.0f; // Double les dégâts si l'élément est vulnérable
        }
        else if (element == this)
        {
            return 0.0f; // Réduit les dégâts à zéro si c'est le même élément
        }
        else
        {
            return 1f; // Dégâts normaux pour les autres éléments
        }
    }
}

Pour ce que ce système fonctionne, il faut ajouter une référence à l'élément dans les tours et dans les ennemis. Il faut d'abord créer les 4 instances d'éléments.

public class GenerateurEnnemis : MonoBehaviour
{
    private Element[] elementsDisponibles;

    private void Start()
    {
        vagueEnCours = false;
        indiceVague = 0;
        instancesEnnemis = new();
        elementsDisponibles = Resources.LoadAll<Element>("Elements");

        StartCoroutine(GenererVagueNiveau());
    }

    /// <summary>
    /// Génère une vague d'ennemis en instanciant le prefab d'ennemi à
    /// l'emplacement de génération et en les initialisant avec la destination.
    /// </summary>
    /// <returns></returns>
    private IEnumerator GenererVague()
    {
        vagueEnCours = true;

        for (int i = 0; i < tailleVague; i++)
        {
            Ennemi ennemi = GenererEnnemi();
            ennemi.Initialiser(emplacementGeneration, destinationEnnemis, $"Ennemi_{numeroSequentielEnnemi++}",
                SelectionnerElement());

            yield return new WaitForSeconds(delaiEntreEnnemi);
        }

        vagueEnCours = false;
    }

    /// <summary>
    /// Retourne une élément aléatoire parmi les éléments disponibles.
    /// </summary>
    /// <returns>Un élément aléatoire parmi les éléments disponibles.</returns>
    private Element SelectionnerElement()
    {
        return elementsDisponibles[UnityEngine.Random.Range(0, elementsDisponibles.Length)];
    }

}

using UnityEngine;
using UnityEngine.AI;

/// <summary>
/// Gère le déplacement et le comportement d'un ennemi dans le jeu. L'ennemi a comme but d'atteindre la base du joueur.
/// Il peut être affecté par des tours et subir des dégâts.
/// </summary>
[RequireComponent(typeof(NavMeshAgent)), RequireComponent(typeof(MeshRenderer))]
public class Ennemi : MonoBehaviour
{    
    // Element de l'ennemi, utilisé pour déterminer les interactions avec les tours et les dégâts reçus.
    private Element element;

    public void Initialiser(GameObject emplacementGeneration, GameObject objectif, string nom, Element element)
    {
        this.objectif = objectif;
        agent.enabled = false;
        this.name = nom;
        this.element = element;

        GetComponent<MeshRenderer>().material = element.Material;

        // ...
    }

    /// <summary>
    /// Inflige des dégâts à l'ennemi et met à jour sa barre de vie. Si les points de vie tombent à 0, l'ennemi est recyclé.
    /// </summary>
    /// <param name="degats">La quantité de dégâts à infliger à l'ennemi</param>
    /// <param name="elementAttaquant">L'élément de l'attaque infligeant les dégâts</param>
    public void RecevoirDegats(float degats, Element elementAttaquant)
    {
        vieActuelle -= degats * element.GetMultiplicateurDegat(elementAttaquant);
        barreVie.MettreAJourProgression(vieActuelle / vieMaximale); 

        if (vieActuelle <= 0)
        {
            ControleurJeu.Instance.GenerateurEnnemis.RecyclerEnnemi(this);
        }
    }
}

Modification des tours¶

On doit commencer par ajouter un script de bouton cliqué pour transmettre l'info à la fabrique. On ajoute aussi un champ de référence à la fabrique de tours dans le contrôleur de jeu.

using UnityEngine;

/// <summary>
/// Projectile lancé par la tour pour attaquer les ennemis.
/// </summary>
public class Projectile : MonoBehaviour
{
    // L'élément du projectile
    private Element element;

    public void Initialiser(Ennemi cible, Vector3 positionInitiale, Quaternion rotationInitiale, float degats, Element element)
    {
        this.cible = cible;
        this.degats = degats;
        transform.SetPositionAndRotation(positionInitiale, rotationInitiale);
        this.element = element;
        GetComponent<MeshRenderer>().material = element.Material;
    }

    private void OnTriggerEnter(Collider autre)
    {
        if (autre.gameObject.TryGetComponent(out Ennemi ennemi) && ennemi == cible)
        {
            // Logique de dégâts à implémenter dans l'ennemi
            cible.RecevoirDegats(degats, element);
            PoolProjectiles.Instance.RecyclerProjectile(this);
        }
    }
}
